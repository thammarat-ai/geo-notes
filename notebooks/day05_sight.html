2. DEM จาก terrarium tiles (ฟรี ไม่ต้อง login)terrarium เข้ารหัสความสูงไว้ใน RGB: h = R·256 + G + B/256 − 32768 (เมตร) ที่ zoom 13 ได้ ~18 ม./พิกเซลเราประกอบ tile 3×3 เป็นหน้าต่าง ~±7 กม. รอบจุด ซึ่งรัศมีพอให้เส้นขอบฟ้า stabilise

import math, urllib.requestfrom rasterio.io import MemoryFileZ=13; SPAN=1; _cache={}def tile(xt,yt):    key=(Z,xt,yt)    if key in _cache: return _cache[key]    h=np.full((256,256),np.nan,float)    url=f'https://s3.amazonaws.com/elevation-tiles-prod/terrarium/{Z}/{xt}/{yt}.png'    try:        data=urllib.request.urlopen(url,timeout=40).read()        with MemoryFile(data) as mem:            with mem.open() as ds: rgb=ds.read()        h=rgb[0].astype('float64')*256+rgb[1]+rgb[2]/256.0-32768.0    except Exception as e:        print('tile fail',e)    _cache[key]=h; return hdef window(lat,lon,span=SPAN):    n=2**Z    gxf=(lon+180)/360*n; gyf=(1-math.asinh(math.tan(math.radians(lat)))/math.pi)/2*n    xt0,yt0=int(gxf),int(gyf); size=2*span+1    BIG=np.full((size*256,size*256),np.nan,float)    for dx in range(-span,span+1):        for dy in range(-span,span+1):            BIG[(dy+span)*256:(dy+span+1)*256,(dx+span)*256:(dx+span+1)*256]=tile(xt0+dx,yt0+dy)    res_m=156543.0339*math.cos(math.radians(lat))/n    opx=int(round((gxf-(xt0-span))*256)); opy=int(round((gyf-(yt0-span))*256))    return BIG,res_m,opx,opyprint('terrarium helper พร้อม')

3. B) Sky-viewshed — เส้นขอบฟ้า 360° และค่า open-skyเดินรังสีครบ 360 ทิศ บันทึกมุมเงยสูงสุดของ terrain (ขอบฟ้า) แล้ว open_sky = เฉลี่ยของมุมที่ว่างเหนือขอบฟ้า / 90°

def sky_viewshed(dem,res_m,opx,opy,eye=2.0):    H,W=dem.shape; horizon=np.zeros(360)    opy=min(max(opy,0),H-1); opx=min(max(opx,0),W-1)    z_obs=float(np.nanmean(dem[max(0,opy-1):opy+2,max(0,opx-1):opx+2]))+eye    maxrange=int(min(H,W)/2)-1    for a in range(360):        th=math.radians(a); cx,cy=math.sin(th),math.cos(th); hmax=-90.0; r=1.0        while r<maxrange:            x=int(round(opx+cx*r)); y=int(round(opy-cy*r))            if 0<=x<W and 0<=y<H:                z=dem[y,x]                if not np.isnan(z):                    ang=math.degrees(math.atan2(z-z_obs,r*res_m))                    if ang>hmax: hmax=ang            r+=1.0        horizon[a]=hmax    open_sky=float(np.mean((90-np.clip(horizon,0,90))/90))    return horizon,z_obs,open_skyrows=[]for s in sites:    dem,res_m,opx,opy=window(s['lat'],s['lon'])    horizon,z_obs,osk=sky_viewshed(dem,res_m,opx,opy)    s.update(z_obs=z_obs,open_sky=osk,mean_horizon=float(np.clip(horizon,0,90).mean()),max_horizon=float(horizon.max()))    np.save('hor_%d.npy'%s['rank'],horizon)    rows.append(s)    print('#%2d open_sky=%.3f meanH=%5.1f maxH=%5.1f z=%5.0fm  %s'%(s['rank'],osk,np.clip(horizon,0,90).mean(),horizon.max(),z_obs,s['name'][:24]))print('\nmean open_sky=%.3f'%np.mean([s['open_sky'] for s in rows]))

6. A) Terrain viewshed จาก 3 ยอดเขาสูงยิงรังสีละเอียด 1,440 ทิศ (ทุก 0.25°) แบบ sub-pixel เพื่อให้ fan ต่อเนื่อง ไม่ใช่จุดกระจาย

def dense_viewshed(dem,res_m,opx,opy,eye=15.0,naz=1440,step=0.5):    H,W=dem.shape; vis=np.zeros((H,W),bool)    opy=min(max(opy,0),H-1); opx=min(max(opx,0),W-1)    z_obs=float(np.nanmean(dem[opy-1:opy+2,opx-1:opx+2]))+eye    maxr=min(H,W)/2-1    for a in range(naz):        th=2*math.pi*a/naz; cx,cy=math.sin(th),math.cos(th); hmax=-90.0; r=1.0        while r<maxr:            x=int(round(opx+cx*r)); y=int(round(opy-cy*r))            if 0<=x<W and 0<=y<H:                z=dem[y,x]                if not np.isnan(z):                    ang=math.degrees(math.atan2(z-z_obs,r*res_m))                    vis[y,x]=ang>=hmax-1e-3                    if ang>hmax: hmax=ang            r+=step    return vis,z_obsS={s['rank']:s for s in rows}; fan_ranks=[2,14,22]fig,axes=plt.subplots(1,3,figsize=(15,5.6),facecolor=DARK_BG)for ax,r in zip(axes,fan_ranks):    s=S[r]; dem,res_m,opx,opy=window(s['lat'],s['lon'],span=2); vis,z_obs=dense_viewshed(dem,res_m,opx,opy)    rad_km=(min(dem.shape)/2)*res_m/1000    ax.imshow(dem,cmap='terrain',origin='upper')    ax.imshow(np.ma.masked_where(~vis,vis.astype(float)),cmap=LinearSegmentedColormap.from_list('v',['#22d3ee','#22d3ee']),alpha=0.72,origin='upper')    ax.plot(opx,opy,marker='*',ms=15,color='white',mec='red',mew=1.2)    ax.set_title('#%d %s\nระดับสายตา %.0f ม. · รัศมี ~%.0f กม.'%(r,s['name'][:18],z_obs,rad_km),color='white',fontsize=10)    ax.set_xticks([]); ax.set_yticks([])fig.suptitle('A) Terrain viewshed จาก 3 ยอดเขาสูง (cyan = พื้นที่ที่สายตาสัมผัสได้) · DEM ~18 ม. · eye 15 ม.',color='white',fontsize=13,fontweight='bold')fig.savefig('day05_fans.png',dpi=150,facecolor=DARK_BG,bbox_inches='tight'); plt.close(fig); print('saved day05_fans.png')
